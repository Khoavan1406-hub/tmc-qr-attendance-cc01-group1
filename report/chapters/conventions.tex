\noindent{\fontsize{15}{18}\selectfont\textbf{Conventions and Content Sources}}\par\vspace{4pt}

\begin{itemize}[leftmargin=36pt, itemsep=2pt, topsep=4pt, parsep=2pt]

  \item \textbf{[Assignment]:} Stated directly in the assignment.

  \item \textbf{[Derived from Assignment]:} Inferred by the team from the assignment; not verbatim.

  \item \textbf{[Team Assumption]:} Condition set by the team where the assignment is silent; confirmation status must be recorded.

  \item \textbf{[Design Proposal]:} Option proposed by the team to meet a requirement; must be evaluated and agreed.

\end{itemize}
